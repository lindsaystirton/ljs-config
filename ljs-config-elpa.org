#+begin_src emacs-lisp
  (defun ljs-config-elpa-install ()
    "Install all ljs-config packages that aren't installed."
    (interactive)
    (dolist (package ljs-config-packages)
      (unless (or (member package package-activated-list)
                  (functionp package))
        (message "Installing %s" (symbol-name package))
        (package-install package))))
  #+end_src

  #+begin_src emacs-lisp
  ;; NOTE: this block was silently dead for a long time (see the
  ;; #+end-src typo fix elsewhere in this file) -- fixing that typo
  ;; let it actually run for the first time, which exposed two real
  ;; bugs of its own: `first' is from the old, unrequired `cl.el' (now
  ;; `car', which is exactly equivalent here), and the auto-install
  ;; call below depends on `ljs-config-packages', a variable that is
  ;; never defined anywhere in this config. Rather than guess at what
  ;; that list should contain, the auto-install trigger is disabled --
  ;; every package actually in use already installs itself correctly
  ;; via its own use-package `:ensure t', so this legacy Emacs
  ;; Starter Kit mechanism is redundant, not load-bearing. `esk-online?'
  ;; is left defined in case it's useful later.
  (when (and (functionp 'network-interface-list)
             (network-interface-list))
    (defun esk-online? ()
      "See if we're online."
      (cl-some (lambda (iface)
                 (unless (equal "lo" (car iface))
                   (member 'up (car (last (network-interface-info
                                             (car iface)))))))
               (network-interface-list))))
  #+end_src
