* Starter Kit Org
Configuration for the eminently useful [[http://orgmode.org/][Org Mode]].

Org-mode is for keeping notes, maintaining ToDo lists, doing project
planning, and authoring with a fast and effective plain-text system.
Org Mode can be used as a very simple folding outliner or as a complex
GTD system or tool for reproducible research and literate programming.

For more information on org-mode check out [[http://orgmode.org/worg/][worg]], a large Org-mode wiki
which is also implemented using Org-mode and [[http://git-scm.com/][git]].

(use-package org :straight nil
  :hook (org-mode . org-mode-setup)
  :config
  (setq org-ellipsis " ▾"
        org-hide-emphasis-markers t)
  ;; Replace list hyphen with dot
  (font-lock-add-keywords 'org-mode
                          '(("^ *\\([-]\\) "
                             (0 (prog1 () (compose-region (match-beginning 1) (match-end 1) "•"))))))

  (defun org-mode-setup ()
    (org-indent-mode 1)
    (variable-pitch-mode 1)
    (auto-fill-mode 0)
    (visual-line-mode 1)))

** PDF-Tools Setup for Org and Org-Roam
#+begin_src emacs-lisp
;; pdf-tools itself is configured once, centrally, in
;; ljs-config-appearance.org (which loads before this file) -- it used
;; to be redeclared here too, with no actual config of its own.

(use-package org-pdftools
  :hook (org-mode . org-pdftools-setup-link)
  :after (org pdf-tools)
  :config
  (setq org-file-apps
        '(("\\.pdf\\'" . emacsclient)))
  (message "org-pdftools loaded"))

(use-package org-noter
  ;; org-pdftools already pulls this in as a dependency (it's what
  ;; teaches org-noter to use pdf-tools instead of doc-view), but
  ;; nothing used to actually configure or bind it -- see the audit.
  ;; Left at org-noter's own defaults for where notes live
  ;; (`org-noter-notes-search-path'/`org-noter-default-notes-file-names',
  ;; ~/Documents/Notes.org) rather than overridden here, since those
  ;; defaults are already portable -- no personal path needed here,
  ;; unlike `citar-bibliography'.
  ;;
  ;; NOTE: deliberately only `:after pdf-tools', not org-pdftools too
  ;; -- org-pdftools itself only loads lazily, on first entering
  ;; org-mode (see its own `:hook' below), so gating this on it meant
  ;; the binding below silently never got installed if a PDF was
  ;; opened before any org buffer was visited in the session. pdf-tools
  ;; itself eagerly requires pdf-view (so pdf-view-mode-map already
  ;; exists once pdf-tools has loaded, right at startup), which is all
  ;; this actually needs.
  :after pdf-tools
  :bind (:map pdf-view-mode-map
         ("C-c C-n" . org-noter))
  :config
  (message "org-noter loaded"))

(use-package org-roam-bibtex
  :after org-roam
  :hook (org-roam-mode . org-roam-bibtex-mode)
  :config
  (setq orb-preformat-keywords
        '("citekey" "title" "url" "author-or-editor" "keywords"))
  (message "org-roam-bibtex loaded"))
#+end_src
